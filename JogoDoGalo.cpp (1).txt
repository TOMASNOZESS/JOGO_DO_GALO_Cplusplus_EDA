//o programa tem que reconhecer que o espaço a mudar é onde consta o '-'
void ttt::inTabuleiro(){
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            tab[i][j] = '-';
        }
    }
}

//design do tabuleiro
void ttt::tabuleiro(){
    cout << "     0       1       2" << endl;
    for (int i = 0; i < 3; ++i) {
        cout << "         |       |       " << endl;
        cout << i << "    ";
        for (int j = 0; j < 3; ++j) {
            if (tab[i][j] == 'X') {
                cout << "\033[38;5;208mX\033[0m";
            } else if (tab[i][j] == 'O') {
                cout << "\033[38;5;33mO\033[0m";
            } else {
                cout << " ";
            }
            if (j < 2) {
                cout << "   |   ";
            }
        }
        cout << endl;
        if (i < 2) {
            cout << "  _______|_______|_______" << endl;
        }
    }
    cout << "         |       |       " << endl;
    cout << "\n" << endl;
}

void ttt::elementar(){
    while(tab[linha][coluna] != '-'){
        linha = rand() % 3;
        coluna = rand() % 3;
    }

    tab[linha][coluna] = marcaC;

}

void ttt::basico(char marca){
    //verificar se há uma jogada para ganhar
    if (verificarJogadaGanhar(marca)) {
        return;
    }
    limparConsole();

    //caso contrário, fazer uma jogada aleatória
    elementar();
    limparConsole();
}

void ttt::medio(char marca){
    //verificar jogada para ganhar
    if (verificarJogadaGanhar(marca)) {
        return;
    }
    limparConsole();

    //verificar jogada para impedir jogador de ganhar
    char adversario = (marca == 'X') ? 'O' : 'X';
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            if (tab[i][j] == '-') {
                tab[i][j] = adversario;
                if (verificarVencedor(adversario)) {
                    tab[i][j] = marca;
                    return;
                }
                tab[i][j] = '-';
            }
        }
    }

    //fazer jogada aleatória caso nada funcione
    elementar();
    limparConsole();
}

void ttt::avancado(char marca){

    if (verificarJogadaGanhar(marca)) {
        return;
    }

    //verificar jogada para impedir jogador de ganhar
    char adversario = (marca == 'X') ? 'O' : 'X';
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            if (tab[i][j] == '-') {
                tab[i][j] = adversario;
                if (verificarVencedor(adversario)) {
                    tab[i][j] = marca;
                    return;
                }
                tab[i][j] = '-';
            }
        }
    }

    //fazer uma jogada para ganhar se possível
    if (verificarJogadaGanhar(marca)) {
        return;
    }

    //fazer uma jogada para evitar a derrota
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            if (tab[i][j] == '-') {
                tab[i][j] = marca;
                if (verificarVencedor(marca)) {
                    return;
                }
                tab[i][j] = '-';
            }
        }
    }

    //se não for possível ganhar ou evitar a derrota, fazer qualquer jogada disponível
    elementar();
    limparConsole();
}

bool ttt::verificarVencedor(char jogador) {
    //verificar linhas e colunas
    for (int i = 0; i < 3; ++i) {
        if (tab[i][0] == jogador && tab[i][1] == jogador && tab[i][2] == jogador) {
            return true;
        }
        if (tab[0][i] == jogador && tab[1][i] == jogador && tab[2][i] == jogador) {
            return true;
        }
    }
    //verificar diagonais
    if (tab[0][0] == jogador && tab[1][1] == jogador && tab[2][2] == jogador) {
        return true;
    }
    if (tab[0][2] == jogador && tab[1][1] == jogador && tab[2][0] == jogador) {
        return true;
    }
    return false;
}

bool ttt::verificarEmpate() {
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            if (tab[i][j] == '-') {
                return false; //há um espaço vazio
            }
        }
    }
    return true; //todos os espaços estão preenchidos
}

//função para verificar se há alguma jogada para ganhar
bool ttt::verificarJogadaGanhar(char marca) {
    //verificar linhas e colunas
    for (int i = 0; i < 3; ++i) {
        // Verificar linhas
        if (tab[i][0] == marca && tab[i][1] == marca && tab[i][2] == '-') {
            tab[i][2] = marca;
            return true;
        }
        if (tab[i][0] == marca && tab[i][2] == marca && tab[i][1] == '-') {
            tab[i][1] = marca;
            return true;
        }
        if (tab[i][1] == marca && tab[i][2] == marca && tab[i][0] == '-') {
            tab[i][0] = marca;
            return true;
        }
        // Verificar colunas
        if (tab[0][i] == marca && tab[1][i] == marca && tab[2][i] == '-') {
            tab[2][i] = marca;
            return true;
        }
        if (tab[0][i] == marca && tab[2][i] == marca && tab[1][i] == '-') {
            tab[1][i] = marca;
            return true;
        }
        if (tab[1][i] == marca && tab[2][i] == marca && tab[0][i] == '-') {
            tab[0][i] = marca;
            return true;
        }
    }
    //verificar diagonais
    if (tab[0][0] == marca && tab[1][1] == marca && tab[2][2] == '-') {
        tab[2][2] = marca;
        return true;
    }
    if (tab[0][0] == marca && tab[2][2] == marca && tab[1][1] == '-') {
        tab[1][1] = marca;
        return true;
    }
    if (tab[1][1] == marca && tab[2][2] == marca && tab[0][0] == '-') {
        tab[0][0] = marca;
        return true;
    }
    if (tab[0][2] == marca && tab[1][1] == marca && tab[2][0] == '-') {
        tab[2][0] = marca;
        return true;
    }
    if (tab[0][2] == marca && tab[2][0] == marca && tab[1][1] == '-') {
        tab[1][1] = marca;
        return true;
    }
    if (tab[1][1] == marca && tab[2][0] == marca && tab[0][2] == '-') {
        tab[0][2] = marca;
        return true;
    }
    return false;
}

void ttt::removerJogador(Jogador jogadores[], int& numJogadores){
    Beep(225,150);
    if (numJogadores == 0) {
        limparConsole();
        cout << "Opcao invalida" << endl;
        return;
    }

    for (int i = 0; i < numJogadores; ++i) {
        cout << i + 1 << ". " << jogadores[i].nome << " - Score: " << jogadores[i].score << endl;
    }
    cout << "\n\n";

    string entrada;
    cout << "Digite o numero ou nome do jogador a ser removido: ";
    //cin >> entrada;
    //cin >> nome;

    getline(cin, entrada);

    if (entrada.empty()) {
        limparConsole();
        Beep(225, 150);
        return;
    }

    int numRemover;
    stringstream ss(entrada);
    if (ss >> numRemover) {
        //verificar se o número do jogador é válido
        if (numRemover < 1 || numRemover > numJogadores) {
            limparConsole();
            Beep(225,150);
            cout << "Jogador invalido" << endl;
            return;
        }
        //remover jogador pelo número
        for (int i = numRemover - 1; i < numJogadores - 1; ++i) {
            jogadores[i] = jogadores[i + 1];
        }
        numJogadores--;
        limparConsole();
        Beep(225,150);
        cout << "Jogador removido com sucesso!" << endl;
    } else {
        //remover jogador pelo nome
        bool encontrado = false;
        for (int i = 0; i < numJogadores; ++i) {
            if (jogadores[i].nome == entrada) {
                encontrado = true;
                for (int j = i; j < numJogadores - 1; ++j) {
                    jogadores[j] = jogadores[j + 1];
                }
                numJogadores--;
                limparConsole();
                Beep(225,150);
                cout << "Jogador removido com sucesso!" << endl;
                break;
            }
        }

        if (!encontrado) {
            limparConsole();
            Beep(225,150);
            cout << "Jogador nao encontrado." << endl;
        }
    }
}

void ttt::escolherJogador(Jogador jogadores[], int numJogadores){
    Beep(225,150);
    //adaptação dos jogador a jogar ao jogo em si
    if (numJogadores == 0) {
        limparConsole();
        cout << "Opcao invalida" << endl;
        return;
    }

    //cout << "Nomes dos jogadores:" << endl;
    for (int i = 0; i < numJogadores; ++i) {
        cout << i + 1 << ". " << jogadores[i].nome << " - Score: " << jogadores[i].score << endl;
    }
    cout << "\n\n";

    string entrada;
    cout << "Escolha o numero ou nome do jogador para jogar: ";
    cin >> entrada;

    int numJogador;
    stringstream ss(entrada);
    if (ss >> numJogador) {
        //verificar se o número do jogador é válido
        if (numJogador < 1 || numJogador > numJogadores) {
            Beep(225,150);
            limparConsole();
            cout << "Jogador invalido" << endl;
            return;
        }
        //selecionar o jogador correspondente ao número inserido
        indiceJogador = numJogador - 1; //ajustar o índice baseado no número
        nomeJogador = jogadores[indiceJogador].nome;
    } else {
        //procurar o jogador pelo nome
        bool encontrado = false;
        for (int i = 0; i < numJogadores; ++i) {
            if (jogadores[i].nome == entrada) {
                encontrado = true;
                indiceJogador = i;
                nomeJogador = jogadores[indiceJogador].nome;
                break;
            }
        }

        if (!encontrado) {
            Beep(225,150);
            limparConsole();
            cout << "Jogador nao encontrado." << endl;
            return;
        }
    }

    limparConsole();
    menu(jogadores, numJogadores);
}
